interface Join<out A, out B> {
    val a: A
    val b: B

    companion object {
        operator fun <A, B> invoke(a: A, b: B): Join<A, B> = object : Join<A, B> {
            override val a: A = a
            override val b: B = b
        }
        operator fun <A, B> invoke(pair: Pair<A, B>): Join<A, B> = invoke(pair.first, pair.second)
    }
}

object CsvReader {
    fun read(path: String, delimiter: String = ",", typeHints: Map<String, TypeMemento> = emptyMap()): Cursor {
        val lines = PlatformFile.readAllLines(path)
        if (lines.isEmpty()) return emptyCursor()

        val headers = lines[0].split(delimiter).map { it.trim() }
        val dataLines = lines.drop(1)

        val types = headers.map { header ->
            typeHints[header] ?: IOMemento.IoString
        }.let { size -> size j { i -> headers[i] j types[i] } }

        val data = dataLines.map { line ->
            val values = line.split(delimiter, limit = headers.size).map { it.trim() }
            headers.size j { i ->
                val rawValue = values.getOrNull(i) ?: ""
                parseValue(rawValue, types[i].b)
            }
        }.let { size -> size j { i -> data[i] as RowVec } }

        return data as Cursor
    }

    private fun parseValue(value: String, type: TypeMemento): Any? {
        return when (type) {

            IOMemento.IoLong -> value.toLongOrNull()
            IOMemento.IoDouble -> value.toDoubleOrNull()
            IOMemento.IoFloat -> value.toFloatOrNull()
            IOMemento.IoBoolean -> value.lowercase() == "true" || value == "1"
            else -> value
        }
    }
